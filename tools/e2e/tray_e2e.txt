# End to end: the status icon, its menu, and hiding a window while its
# connection stays alive, against the loopback SSH fixture on port 2222.
#
#   (cd tools/e2esshd && go build -o /tmp/e2esshd . && /tmp/e2esshd 2222) &
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/tray_e2e.txt
#
# Every "windows" line is a check to read: hidden=true is a window the user
# closed that went to the tray. The menu is the system's, so it is read with
# statusmenu and used with statusselect, not clicked.

# One real session to list.
click "Search"
type tester@127.0.0.1:2222
click "Connect"
waitfor "Trust" 3000
click "Trust and Save"
waitfor "Password" 3000
type secret
key enter
waitfor "tester@e2e" 8000

# The application put one icon in the status area, and it has a menu.
statusitems

# The menu lists the session as connected, then the fixed entries.
statusmenu 1

# A menu entry opens a shell on this computer, and the count follows.
statusselect 1 4
waitfor "This Computer" 4000
statusmenu 1

# Closing the window hides it; the connection goes on. Prove that by running a
# command in the hidden window's shell.
window 1
requestclose 1
windows
type echo alive-$((6*7))
key enter
waitfor "alive-42" 4000

# Show Motty (entry 6 now that two sessions are listed) brings the window back.
statusselect 1 6
windows

# A primary press, which a Linux tray reports as an activation, brings it back too.
requestclose 1
statusclick 1
windows

# Close it again, then choose the session: the window comes back at it.
requestclose 1
windows
statusselect 1 2
windows

# Quit from the menu (entry 8) ends the application and its connections.
statusselect 1 8
