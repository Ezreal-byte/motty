# End to end: a split of a connected host borrows its SSH connection. Against the
# loopback SSH fixture on port 2222 (see tray_e2e.txt for starting it).
#
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/share_e2e.txt
#
# The password is typed once. The split needs no host-key prompt, no password and
# no new handshake, and a shell answers.

click "Search"
type tester@127.0.0.1:2222
click "Connect"
waitfor "Trust" 3000
click "Trust and Save"
waitfor "Password" 3000
type secret
key enter
waitfor "tester@e2e" 8000

# The split is a second shell on the same connection.
click "Split Down"
waitfor "tester@e2e" 4000
type echo second-$((6*7))
key enter
waitfor "second-42" 4000
texts
