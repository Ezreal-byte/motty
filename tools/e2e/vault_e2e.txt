# End to end: a password the user asks to save is kept, and the next connection
# to the same account needs no password dialog. Against the loopback SSH fixture
# on port 2222 (see tray_e2e.txt for starting it).
#
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/vault_e2e.txt
#
# The headless build keeps secrets in memory, so this is the vault, its encrypted
# file and the prompts, not the operating system's keychain.

click "Search"
type tester@127.0.0.1:2222
click "Connect"
waitfor "Trust" 3000
click "Trust and Save"
waitfor "Password" 3000
type secret
click "Save password"
click "Connect"
waitfor "tester@e2e" 8000

# Close the tab, then connect again: no Trust, no Password, straight to the shell.
key cmd+w
click "Search"
key cmd+a
type tester@127.0.0.1:2222
click "Connect"
waitfor "tester@e2e" 8000
texts
