#
#   go build -tags vivahead -o /tmp/motty-head-sshsync ./cmd/motty
#   home=$(mktemp -d); mkdir $home/.ssh; cp some_config $home/.ssh/config
#   MOTTY_LANG=en MOTTY_DATA_DIR=$(mktemp -d) MOTTY_HEAD=/tmp/motty-head-sshsync \
#     python3 tools/headless_drive.py $home tools/e2e/ssh_config_sync_e2e.txt
#
# Afterwards <home>/.ssh/config must hold every line it began with, plus a block
# of "# --- motty: begin synced hosts" with the host made below, and the hosts
