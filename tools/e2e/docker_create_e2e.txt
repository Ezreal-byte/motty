# End to end: pulling an image and creating a volume and a network from the remote
# tools dialog, against the loopback SSH fixture on port 2222 started with
# E2E_ALLOW_EXEC=1 so that the fixed commands run for real on this computer's
# Docker (see remote_tools_e2e.txt for the setup). Chinese labels; the language of
# the machine decides them.
#
# The volume motty-e2e-vol and the network motty-e2e-net are created and stay
# until removed: `docker volume rm motty-e2e-vol; docker network rm motty-e2e-net`.
# The pull uses redis:alpine, so it needs the image already here or the network.

click 462 68
type tester@127.0.0.1:2222
click "连接"
waitfor "信任" 3000
click "信任并保存"
waitfor "密码" 3000
type secret
key enter
waitfor "tester@e2e" 8000
settle 800

click "命令面板"
type 进程与容器
waitfor "进程与容器" 2000
key enter
waitfor "远程工具" 3000
settle 2000

click "卷"
settle 1500
click "新建"
waitfor "新建卷" 2000
type motty-e2e-vol
shot /tmp/docker-new-volume.png
key enter
settle 2500
waitfor "motty-e2e-vol" 4000

click "网络"
settle 1500
click "新建"
waitfor "新建网络" 2000
type motty-e2e-net
key enter
settle 2500
waitfor "motty-e2e-net" 4000
shot /tmp/docker-networks.png

click "镜像"
settle 1500
click "拉取镜像"
waitfor "拉取镜像" 2000
type redis:alpine
shot /tmp/docker-pull.png
key enter
waitfor "redis:alpine" 15000
settle 4000
texts "redis"
