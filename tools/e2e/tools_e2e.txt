# End to end: broadcast typing, a snippet from the command palette and a terminal
# color scheme, against the loopback SSH fixture on port 2222 (see tray_e2e.txt).
#
#   python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/tools_e2e.txt
#
# After the broadcast line the two panes each print "pane-42", so the last texts
# output must list it twice. The scheme step reads the terminal background back
# through a shot the reader opens; the color itself is checked by the unit tests.

click "Search"
type tester@127.0.0.1:2222
click "Connect"
waitfor "Trust" 3000
click "Trust and Save"
waitfor "Password" 3000
type secret
key enter
waitfor "tester@e2e" 8000

click "Split Down"
waitfor "Using the connection of" 4000
waitfor "tester@e2e" 4000
# The resource monitor's button appears a moment after connecting and moves the
# others; let the bar settle before pressing one of them.
settle 800
click "Type into every pane of this tab"
click "Terminal tester@127.0.0.1:2222"
type echo pane-$((6*7))
key enter
wait 1500
texts

# Turn it off: a second line reaches only the pane that has the keyboard.
click "Stop typing into every pane"
click "Terminal tester@127.0.0.1:2222"
type echo solo-$((6*7))
key enter
waitfor "solo-42" 4000
texts
