# End to end: a host with "Forward ssh-agent" on lets the remote side reach the
# local ssh-agent. Needs the loopback fixture with agent forwarding on, and a
# real ssh-agent holding one key whose socket the headless binary inherits:
#
#   go build -tags vivahead -o /tmp/motty-head ./cmd/motty
#   go build -o /tmp/motty-e2esshd ./tools/e2esshd
#   eval "$(ssh-agent -s)"; ssh-add /tmp/agent-forward-key
#   E2E_AGENTFWD=1 SSH_AUTH_SOCK=$SSH_AUTH_SOCK MOTTY_LANG=en /tmp/motty-e2esshd 2244 &
#   MOTTY_LANG=en python3 tools/headless_drive.py "$(mktemp -d)" tools/e2e/agent_forward_e2e.txt
#
# The shell prompt is followed by the AGENTFWD line the server printed after
# querying the forwarded agent, naming the key's comment.
